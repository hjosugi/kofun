fn classify(read value: Int) -> Text {
  let mut total = value + 2 * 3
  if total >= 9 {
    return "large"
  } else {
    return "small"
  }
}

fn main() {
  let values = 1 .. 8
  let answer = values
  |> map(fn(x: Int) => x * x)
  |> sum()
  print(answer)
}
